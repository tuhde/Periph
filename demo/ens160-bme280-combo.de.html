SparkFun Umwelt-Kombisensor · ENS160 + BME280 · Qwiic

Zwei Sensoren,
ein Kabel.

Dieses Board vereint zwei Sensoren auf einer einzigen, nur 25×25 mm kleinen Platine. Der BME280 von Bosch misst Temperatur, Luftfeuchtigkeit und Luftdruck. Der ENS160 von ScioSense bewertet die Luftqualität – also wie viele flüchtige Stoffe (VOC) und wie viel CO2 sich in der Raumluft befinden. Angeschlossen wird alles über ein einziges Qwiic-Steckkabel, ganz ohne Löten. Damit eignet sich das Board für Wetterstationen, Raumklima-Monitore oder eine automatische Lüftungssteuerung, die selbst erkennt, wann gelüftet werden sollte.

Für Entwickler: Beide Chips werden von unserer Open-Source-Bibliothek periph vollständig unterstützt – fertige Beispielprogramme gibt es in sechs Programmiersprachen und für die gängigsten Plattformen, vom Arduino bis zum Raspberry Pi.

Zwei Sensoren, zwei Aufgaben

Jeder Sensor arbeitet unabhängig vom anderen – mit eigener Adresse, eigenen Werkseinstellungen und eigener Messgenauigkeit.

Fertiger Code in sechs Sprachen

Dasselbe Board, sechsmal ausgelesen. Hier zu sehen: die Linux-Variante jedes Treibers.

Jeder Treiber legt die Geräteadresse schon beim Verbindungsaufbau fest – es gibt also keinen einzelnen Anschluss, der für beide Chips gleichzeitig gilt. Jedes Beispiel unten öffnet daher denselben I²C-Bus zweimal, einmal je Adresse. Das ist unter Linux normal: /dev/i2c-1 lässt sich problemlos von zwei unabhängigen Verbindungen gleichzeitig öffnen. 

from smbus2 import SMBus
from periph.connection.i2c_linux import I2CConnection
from periph.chips.gas.ens160 import ENS160Minimal
from periph.chips.environmental.bme280 import BME280Minimal
import time

bus = SMBus(1)                                # ein gemeinsamer Bus-Handle
ens = ENS160Minimal(I2CConnection(bus, 0x52))  # Luftqualität, Adresse 0x52
bme = BME280Minimal(I2CConnection(bus, 0x76))  # Klimadaten, Adresse 0x76

while True:
    try:
        air = ens.read_air_quality()          # {'aqi', 'tvoc_ppb', 'eco2_ppm'}
    except RuntimeError:
        time.sleep(1)                          # Sensor wärmt noch auf
        continue
    t, p, h = bme.temperature(), bme.pressure(), bme.humidity()
    print(f"AQI={air['aqi']} TVOC={air['tvoc_ppb']}ppb eCO2={air['eco2_ppm']}ppm "
          f"| {t:.1f}C {p:.1f}hPa {h:.1f}%RH")
    time.sleep(1)
#include <cstdio>
#include <unistd.h>
#include "I2CConnectionLinux.h"
#include "ENS160.h"
#include "BME280.h"

int main() {
    I2CConnectionLinux ens_conn(1, 0x52);   // Luftqualität, Bus 1, Adresse 0x52
    I2CConnectionLinux bme_conn(1, 0x76);   // Klimadaten, Bus 1, Adresse 0x76

    ENS160Minimal ens(ens_conn);
    BME280Minimal bme(bme_conn);

    while (true) {
        uint8_t aqi; float tvoc_ppb, eco2_ppm;
        if (ens.read_air_quality(aqi, tvoc_ppb, eco2_ppm)) {
            float t = bme.temperature();
            float p = bme.pressure();
            float h = bme.humidity();
            printf("AQI=%u TVOC=%.1fppb eCO2=%.1fppm | %.2fC %.2fhPa %.2f%%RH\n",
                   aqi, tvoc_ppb, eco2_ppm, t, p, h);
        }
        usleep(1000000);
    }
}

const { ENS160Minimal } = require('periph/chips/gas/ens160');
const { BME280Minimal } = require('periph/chips/environmental/bme280');

const bme = new BME280Minimal(new I2CConnection(1, 0x76));  // Klimadaten

(async () => {
  while (true) {
    try {
      const air = await ens.readAirQuality();     // {aqi, tvocPpb, eco2Ppm}
      const t = await bme.temperature();
      const p = await bme.pressure();
      const h = await bme.humidity();
      console.log(`AQI=${air.aqi} TVOC=${air.tvocPpb}ppb eCO2=${air.eco2Ppm}ppm `
        + `| ${t.toFixed(1)}C ${p.toFixed(1)}hPa ${h.toFixed(1)}%RH`);
    } catch (e) { /* ENS160 wärmt noch auf */ }
    await new Promise(r => setTimeout(r, 1000));
  }
})();
use linux_embedded_hal::I2cdev;
use periph::chips::gas::Ens160Minimal;
use periph::chips::environmental::Bme280Minimal;

fn main() {
    let ens_dev = I2cdev::new("/dev/i2c-1").expect("Bus öffnen");
    let bme_dev = I2cdev::new("/dev/i2c-1").expect("Bus öffnen");
    let mut ens = Ens160Minimal::new(ens_dev, 0x52).expect("ENS160 init");
    let mut bme = Bme280Minimal::new(bme_dev, 0x76, false).expect("BME280 init");

    loop {
        let status = ens.wait_for_new_data(2000).unwrap();
        if (status >> 2) & 0x03 != 0 { continue; }  // wärmt noch auf

        let (aqi, tvoc_ppb, eco2_ppm) = ens.read_air_quality().unwrap();
        let t = bme.temperature().unwrap();
        let p = bme.pressure().unwrap();
        let h = bme.humidity().unwrap();
        println!("AQI={aqi} TVOC={tvoc_ppb}ppb eCO2={eco2_ppm}ppm | {t:.1}C {p:.1}hPa {h:.1}%RH");
        std::thread::sleep(std::time::Duration::from_secs(1));
    }
}
